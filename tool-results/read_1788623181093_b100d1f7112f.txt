    43→// Static FAQ data — used in static export mode (GitHub Pages) where API routes

    45→type StaticFaq = { id: string; question: string; answer: string; keywords?: string; category?: string };
    46→let staticFaqs: StaticFaq[] = [];
    47→
    48→/** Client-side FAQ matching for static export mode (no API). */
    49→function matchStaticFaq(visitorQuestion: string): StaticFaq | null {
    50→  if (staticFaqs.length === 0) return null;
    51→  const q = visitorQuestion.toLowerCase().replace(/[^\w\s]/g, " ").replace(/\s+/g, " ").trim();
    52→  const qTokens = new Set(q.split(" ").filter((t) => t.length > 2));
    53→  let best: StaticFaq | null = null;
    54→  let bestScore = 0;
    55→  for (const f of staticFaqs) {
    56→    const fQ = f.question.toLowerCase();
    57→    const keywords = (f.keywords ?? "").split(",").map((k) => k.trim().toLowerCase()).filter(Boolean);
    58→    let score = 0;
    59→    if (q.includes(fQ) || fQ.includes(q)) score += 6;
    60→    for (const k of keywords) { if (q.includes(k)) score += 2; if (qTokens.has(k)) score += 1; }
    61→    const fTokens = new Set(fQ.split(" ").filter((t) => t.length > 2));
    62→    for (const t of qTokens) { if (fTokens.has(t)) score += 0.5; }
    63→    if (score > bestScore) { bestScore = score; best = f; }
    64→  }
    65→  return bestScore >= 2 ? best : null;
    66→}
